// @vitest-environment jsdom
import { expect, it, vi } from "vitest";
vi.mock("@/lib/demo/demo-session", () => ({ isDemoModeActive: () => false }));
it("a resident read must not reuse the same user's manager response", async () => {
  const { setPortalSessionViewer } = await import("@/lib/auth/portal-session-gate");
  const { syncManagerApplicationsFromServer } = await import("@/lib/manager-applications-storage");
  setPortalSessionViewer("review-multi-role-user");
  const fetchMock = vi.fn().mockResolvedValue({ ok: true, status: 200, json: async () => ({ rows: [] }) });
  vi.stubGlobal("fetch", fetchMock);
  await syncManagerApplicationsFromServer({ managerUserId: "review-multi-role-user" });
  await syncManagerApplicationsFromServer({ selfScope: true });
  expect(fetchMock.mock.calls.map(([url]) => url)).toEqual([
    "/api/manager-applications",
    "/api/manager-applications?scope=self",
  ]);
});
it("a resident service read must not reuse a fresh manager-portal response", async () => {
  const { syncServiceRequestsFromServer } = await import("@/lib/service-requests-storage");
  document.cookie = "axis_active_portal=manager; path=/";
  const fetchMock = vi.fn().mockResolvedValue({ ok: true, status: 200, json: async () => ({ rows: [] }) });
  vi.stubGlobal("fetch", fetchMock);
  await syncServiceRequestsFromServer();
  document.cookie = "axis_active_portal=resident; path=/";
  await syncServiceRequestsFromServer();
  expect(fetchMock).toHaveBeenCalledTimes(2);
});
